/**\n * Dice class - matches Android version functionality\n */\nclass Dice {\n    constructor(numSides = 6) {\n        this.numSides = numSides;\n    }\n    \n    roll() {\n        return Math.floor(Math.random() * this.numSides) + 1;\n    }\n}\n\n/**\n * DiceApp class - main application logic\n */\nclass DiceApp {\n    constructor() {\n        this.dice = new Dice(6);\n document.getElementById('rollButton');\n        const diceNumber = document.getElementById('diceNumber');\n        \n        // Disable button during animation\n        rollButton.disabled = true;\n        rollButton.classList.add('rolling');\n        \n        // Animation effect\n        await this.animateRoll();\n        \n        // Get actual roll result\n        const result = this.dice.roll();\n        \n        // Update display\n        diceNumber.textContent = result;\n        diceNumber.classList.add('result-appear');\n        \n        // Update statistics\n        this.updateStats(result);\n        \n        // Re-enable button\n        setTimeout(() => {\n            rollButton.disabled = false;\n            rollButton.classList.remove('rolling');\n            diceNumber.classList.remove('result-appear');\n            this.isAnimating = false;\n        }, 500);\n    }\n    \n    async animateRoll() {\n        const diceNumber = document.getElementById('diceNumber');\n        const animationDuration = 1000;\n        const frameRate = 100;\n        const frames = animationDuration / frameRate;\n        \n        for (let i = 0; i < frames; i++) {\n            const randomNum = Math.floor(Math.random() * 6) + 1;\n            diceNumber.textContent = randomNum;\n            await new Promise(resolve => setTimeout(resolve,    if (this.rollHistory.length > 100) {\n            this.rollHistory = this.rollHistory.slice(-100);\n        }\n        \n        const average = this.rollHistory.reduce((a, b) => a + b, 0) / this.rollHistory.length;\n        \n        // Update UI\n        document.getElementById('totalRolls').textContent = this.totalRolls;\n        document.getElementById('lastRoll').textContent = result;\n        document.getElementById('averageRoll').textContent = average.toFixed(2);\n        \n        // Save to localStorage\n        this.saveStats();\n    }\n    \n    saveStats() {\n    loadStats() {\n        const saved = localStorage.getItem('diceRollerStats');\n        if (saved) {\n            const stats = JSON.parse(saved);\n            this.totalRolls = stats.totalRolls || 0;\n            this.rollHistory = stats.rollHistory || [];\n            \n            if (this.rollHistory.length > 0) {\n                const lastRoll = this.rollHistory[this.rollHistory.length - 1];\n                const average = this.rollHistory.reduce((a, b) => a + b, 0) / this.rollHistory.length;\n                \n                document.getElementById('totalRolls').textContent = this.totalRolls;\n                document.getElementById('lastRoll').textContent = lastRoll;\n                document.getElementById('averageRoll').textContent = average.toFixed(2);\n            }\n        }\n    }\n    \n    toggleStats() {\n        const statsContainer = document.getElementById('statsContainer');\n        const toggleButton = document.getElementById('toggleStats');\n        \n        if (statsContainer.style.display === 'none') {\n            statsContainer.style.display = 'block';\n            toggleButton.textContent = 'Hide Statistics';\n        } else {\n            statsContainer.style.display = 'none';\n            toggleButton.textContent = 'Show Statistics';\n        }\n    }\n    \n    async setupServiceWorker() {\n        if ('serviceWorker' in navigator) {\n            try {\n                await navigator.serviceWorker.register('/sw.js');\n                console.log('Service Worker registered successfully');\n            } catch (error) {\n                console.log('Service Worker registration failed:', error);\n            }\n        }\n    }\n}\n\n// Initialize app when DOM is loaded\nif (typeof window !== 'undefined') {\n    window.diceApp = new DiceApp();\n}\n```
