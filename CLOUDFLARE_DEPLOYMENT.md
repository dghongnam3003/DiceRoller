API Implementation\n\n**src/index.js:**\n```javascript\n// Dice Roller API using Cloudflare Workers\n\nclass DiceAPI {\n  constructor() {\n    this.routes = new Map([\n      ['GET /api/roll', this.handleRoll.bind(this)],\n      ['GET /api/history', this.handleHistory.bind(this)],\n      ['POST /api/history', this.handleSaveHistory.bind(this)],\n      ['GET /api/stats', this.handleStats.bind(this)],\n    ]);\n  }\n\n  async handleRequest(request) {\n    const url = new URL(request.url);\n    const path = url.pathname;\n    const method = request.method;\n    const routeKey = `${method} ${path}`;\n\n    // CORS headers\n    const corsHeaders = {\n      'Access-Control-Allow-Origin': '*',\n      'Access-Control-Allow-Methods': 'GET, POST, OPTIONS',\n      'Access-Control-Allow-Headers': 'Content-Type',\n    };\n\n    // Handle CORS preflight\n    if (method === 'OPTIONS') {\n      return new Response(null, { headers: corsHeaders });\n    }\n\n    // Route to handler\n    const handler = this.routes.get(routeKey);\n    if (handler) {\n      try {\n        const response = await handler(request, url);\n        // Add CORS headers to response\n        Object.entries(corsHeaders).forEach(([key, value]) => {\n          response.headers.set(key, value);\n        });\n        return response;\n      } catch (error) {\n        return this.errorResponse(error.message, 500, corsHeaders);\n      }\n    }\n\n    return this.errorResponse('Not Found', 404, corsHeaders);\n  }\n\n  async handleRoll(request, url) {\n    const params = url.searchParams;\n    const sides = parseInt(params.get('sides')) || 6;\n    const count = Math.min(parseInt(params.get('count')) || 1, 10);\n\n    // Validate input\n    if (sides < 2 || sides > 100) {\n      return this.errorResponse('Invalid number of sides (2-100)', 400);\n    }\n\n    const rolls = [];\n    for (let i = 0; i < count; i++) {\n      rolls.push({\n        result: Math.floor(Math.random() * sides) + 1,\n        sides: sides,\n        timestamp: Date.now()\n      });\n    }\n\n    // Track usage stats (optional)\n    await this.trackUsage(request, { sides, count });\n\n    return new Response(JSON.stringify({\n      success: true,\n      rolls: rolls,\n      total: rolls.reduce((sum, roll) => sum + roll.result, 0)\n    }), {\n      headers: { 'Content-Type': 'application/json' }\n    });\n  }\n\n  async handleHistory(request, url) {\n    // Get user's roll history from KV storage\n    const userId = request.headers.get('X-User-ID') || 'anonymous';\n    const history = await DICE_HISTORY.get(`history:${userId}`);\n\n    return new Response(JSON.stringify({\n      success: true,\n      history: history ? JSON.parse(history) : []\n    }), {\n      headers: { 'Content-Type': 'application/json' }\n    });\n  }\n\n  async handleSaveHistory(request, url) {\n    const userId = request.headers.get('X-User-ID') || 'anonymous';\n    const { rolls } = await request.json();\n\n    // Validate and save history\n    if (Array.isArray(rolls) && rolls.length <= 100) {\n      await DICE_HISTORY.put(\n        `history:${userId}`,\n        JSON.stringify(rolls),\n        { expirationTtl: 86400 * 30 } // 30 days\n      );\n    }\n\n    return new Response(JSON.stringify({ success: true }), {\n      headers: { 'Content-Type': 'application/json' }\n    });\n  }\n\n  async handleStats(request, url) {\n    // Get aggregated statistics\n    const stats = await DICE_STATS.get('global-stats');\n    const parsed = stats ? JSON.parse(stats) : {\n      totalRolls: 0,\n      uniqueUsers: 0,\n      popularSides: { 6: 0 }\n    };\n\n    return new Response(JSON.stringify({\n      success: true,\n      stats: parsed\n    }), {\n      headers: { 'Content-Type': 'application/json' }\n    });\n  }\n\n  async trackUsage(request, { sides, count }) {\n    // Simple usage tracking\n    try {\n      const stats = await DICE_STATS.get('global-stats');\n      const current = stats ? JSON.parse(stats) : {\n        totalRolls: 0,\n        popularSides: {}\n      };\n\n      current.totalRolls += count;\n      current.popularSides[sides] = (current.popularSides[sides] || 0) + count;\n\n      await DICE_STATS.put('global-stats', JSON.stringify(current));\n    } catch (error) {\n      console.error('Failed to track usage:', error);\n    }\n  }\n\n  errorResponse(message, status = 400, headers = {}) {\n    return new Response(JSON.stringify({\n      success: false,\n      error: message\n    }), {\n      status,\n      headers: {\n        'Content-Type': 'application/json',\n        ...headers\n      }\n    });\n  }\n}\n\n// Worker event listener\naddEventListener('fetch', event => {\n  const api = new DiceAPI();\n  event.respondWith(api.handleRequest(event.request));\n});\n```\n\n#### 3.
